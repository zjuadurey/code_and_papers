6: def preview(matrix: list[list[int]], slot_count: int) -> list[int]:
7:     # First available slot in session order. A blocked preview is inconclusive.
8:     n = len(matrix)
9:     assigned = [-1] * n
10:     v = 0
11:     while v < n:
12:         used = set()
13:         u = 0
14:         while u < n:
15:             if matrix[v][u] == 1 and assigned[u] != -1:
16:                 used.add(assigned[u])
17:             u += 1
18:         slot = 0
19:         while slot < slot_count and slot in used:
20:             slot += 1
21:         if slot == slot_count:
22:             raise ValueError("preview blocked")
23:         assigned[v] = slot
24:         v += 1
25:     return assigned
26:

28: def complete(neighbors: dict[int, list[int]], slot_count: int) -> list[int]:
29:     n = len(neighbors)
30:     assigned = [-1] * n
31: 
32:     def available(v: int, slot: int) -> bool:
33:         idx = 0
34:         linked = neighbors[v]
35:         while idx < len(linked):
36:             u = linked[idx]
37:             if assigned[u] == slot:
38:                 return False
39:             idx += 1
40:         return True
41: 
42:     def place(v: int) -> bool:
43:         if v == n:
44:             return True
45:         slot = 0
46:         while slot < slot_count:
47:             if available(v, slot):
48:                 assigned[v] = slot
49:                 if place(v + 1):
50:                     return True
51:                 assigned[v] = -1
52:             slot += 1
53:         return False
54: 
55:     if not place(0):
56:         raise ValueError("no arrangement")

6: def prepare(request: Any) -> tuple[list[str], list[dict], list[str | None], str]:
7:     if not isinstance(request, dict) or set(request) != {"slots", "sessions", "current", "mode"}:

11:             or len(set(slots)) != len(slots)):
12:         raise ValueError("slots must be distinct names")
13:     if not isinstance(sessions, list) or mode not in ("inspect", "complete"):
14:         raise ValueError("invalid sessions or mode")
15:     ids = set()

18:             raise ValueError("invalid session")
19:         name, participants = session["session_id"], session["participants"]
20:         if not isinstance(name, str) or not name or name in ids:
21:             raise ValueError("session IDs must be unique")
22:         ids.add(name)
23:         if (not isinstance(participants, list) or any(not isinstance(p, str) or not p for p in participants)
24:                 or len(set(participants)) != len(participants)):
25:             raise ValueError("participants must be distinct names")

27:             or any(s is not None and (not isinstance(s, str) or s not in slots) for s in current)):

29:     return list(slots), sessions, list(current), mode

32: def relations(sessions: list[dict]) -> tuple[list[list[int]], dict[int, list[int]]]:
33:     matrix = [[0] * len(sessions) for _ in sessions]
34:     for i, left in enumerate(sessions):
35:         for j in range(i + 1, len(sessions)):
36:             if set(left["participants"]) & set(sessions[j]["participants"]):
37:                 matrix[i][j] = matrix[j][i] = 1
