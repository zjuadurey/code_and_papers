33:         matrix[v][u] += weight

74: def maxcut_bruteforce(adj):
75:     n=len(adj)
76:     best=-1
77:     best_part=None
78:     for mask in range(1<<n):
79:         left=[i for i in range(n) if (mask>>i)&1]
80:         right=[i for i in range(n) if not (mask>>i)&1]
