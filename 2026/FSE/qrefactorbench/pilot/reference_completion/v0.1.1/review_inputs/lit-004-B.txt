6: def relation(names: list[str], latest: dict[frozenset[str], bool]) -> list[list[bool]]:
7:     return [[i != j and latest.get(frozenset((left, right))) is True for j, right in enumerate(names)]
8:             for i, left in enumerate(names)]

11: def compatible(indices: list[int], matrix: list[list[bool]]) -> bool:
12:     return all(matrix[u][v] for u in indices for v in indices if u != v)

15: def preview(matrix: list[list[bool]]) -> list[int]:
16:     selected = []
17:     for node in range(len(matrix)):
18:         if all(matrix[node][neighbor] for neighbor in selected):
19:             selected.append(node)
20:     return selected

24:     ordered = [v["version_id"] for v in versions if v["version_id"] in selected]
25:     inactive = [v["version_id"] for v in versions if v["version_id"] in selected and not v["active"]]
