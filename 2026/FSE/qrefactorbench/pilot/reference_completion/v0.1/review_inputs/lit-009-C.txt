FILE kernel.py
1: """Dense elimination with partial pivoting and an HPL-style residual diagnostic.
2: 
3: New small Python implementation of the documented problem, not an HPL port/rating.
4: """
5: import math
6: import sys
7: 
8: 
9: def solve(matrix, rhs):
10:     n = len(rhs)
11:     rows = [list(row) + [b] for row, b in zip(matrix, rhs)]
12:     for col in range(n):
13:         pivot = max(range(col, n), key=lambda i: abs(rows[i][col]))
14:         if rows[pivot][col] == 0:
15:             raise ValueError("singular system")
16:         rows[col], rows[pivot] = rows[pivot], rows[col]
17:         for i in range(col + 1, n):
18:             factor = rows[i][col] / rows[col][col]
19:             for j in range(col, n + 1):
20:                 rows[i][j] -= factor * rows[col][j]
21:     x = [0.0] * n
22:     for i in range(n - 1, -1, -1):
23:         x[i] = (rows[i][n] - sum(rows[i][j] * x[j] for j in range(i + 1, n))) / rows[i][i]
24:     if any(not math.isfinite(v) for v in x):
25:         raise ValueError("numerical overflow")
26:     return x
27: 
28: 
29: def residual(matrix, rhs, x):
30:     errors = [sum(a * v for a, v in zip(row, x)) - b for row, b in zip(matrix, rhs)]
31:     norm = max(map(abs, errors), default=0.0)
32:     a_norm = max((sum(map(abs, row)) for row in matrix), default=0.0)
33:     denominator = sys.float_info.epsilon * (a_norm * max(map(abs, x), default=0.0)
34:                    + max(map(abs, rhs), default=0.0)) * len(rhs)
35:     if any(not math.isfinite(v) for v in errors + [norm, denominator]):
36:         raise ValueError("numerical overflow")
37:     scaled = norm / denominator if denominator else (0.0 if norm == 0 else None)
38:     return {"vector": errors, "infinity_norm": norm, "scaled_backward_error": scaled}
