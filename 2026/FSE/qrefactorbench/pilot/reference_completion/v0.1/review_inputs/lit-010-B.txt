1: """Small unpreconditioned CG path derived from HPCG's documented recurrence.
2: 
3: HPCG source: University of Tennessee/UT-Battelle/Sandia, BSD-3-Clause; NOTICE.txt.
4: No MPI, multigrid, performance rating or C++ translation system is implemented.
5: """
6: import math
7: 
8: 
9: def dot(a, b):
10:     return sum(x * y for x, y in zip(a, b))
11: 
12: 
13: def apply(rows, x):
14:     return [sum(value * x[j] for j, value in row) for row in rows]
15: 
16: 
17: def advance(rows, rhs, current, steps, tolerance):
18:     x = list(current)
19:     r = [b - v for b, v in zip(rhs, apply(rows, x))]
20:     initial = math.sqrt(dot(r, r))
21:     if not math.isfinite(initial):
22:         raise ValueError("numerical overflow")
23:     norm, trace, p, previous = initial, [], list(r), None
24:     while len(trace) < steps and initial and norm / initial > tolerance:
25:         rr = dot(r, r)
26:         if previous is not None:
27:             p = [v + rr / previous * old for v, old in zip(r, p)]
28:         ap = apply(rows, p)
29:         denominator = dot(p, ap)
30:         if denominator <= 0 or not math.isfinite(denominator):
31:             raise ValueError("numerical breakdown")
32:         alpha = rr / denominator
33:         x = [v + alpha * direction for v, direction in zip(x, p)]
34:         r = [v - alpha * av for v, av in zip(r, ap)]
35:         previous = rr
36:         norm = math.sqrt(dot(r, r))
37:         if not math.isfinite(norm) or any(not math.isfinite(v) for v in x):
38:             raise ValueError("numerical overflow")
39:         trace.append({"iteration": len(trace) + 1, "residual_norm": norm})
40:     return x, initial, trace
